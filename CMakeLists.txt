# Blaise on any Native backend. `make` remains the
# local entry point; this file is what the platform SDKs build.
cmake_minimum_required(VERSION 3.20)
project(blaise LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BLAISE_VERSION "0.1.0" CACHE STRING "Application version")
if(NOT NATIVE_ROOT)
    get_filename_component(NATIVE_ROOT "${CMAKE_SOURCE_DIR}/../native" ABSOLUTE)
endif()
set(BUILD_TESTING OFF CACHE BOOL "Native standalone tests" FORCE)
add_subdirectory(src/ui native-dependency)

set(libraries view compose filter addr mail net prefs mime util)
add_library(blaise_options INTERFACE)
target_include_directories(blaise_options INTERFACE include)
foreach(name ${libraries})
    target_include_directories(blaise_options INTERFACE "lib/${name}/include")
endforeach()
target_compile_options(blaise_options INTERFACE
    $<$<CXX_COMPILER_ID:GNU>:-Wall;-Wextra;-pedantic>)
target_compile_definitions(blaise_options INTERFACE
    BLAISE_VERSION="${BLAISE_VERSION}")

# TLS is optional: without it a connection that asks for TLS fails with a
# clear message. See docs/notes/TLS.md.
find_package(OpenSSL QUIET)
if(OPENSSL_FOUND)
    target_compile_definitions(blaise_options INTERFACE BLAISE_TLS_OPENSSL)
    target_link_libraries(blaise_options INTERFACE OpenSSL::SSL OpenSSL::Crypto)
endif()
find_package(Threads REQUIRED)
target_link_libraries(blaise_options INTERFACE Threads::Threads)
# Sockets are in the C library on Linux, but not on these two.
if(WIN32)
    target_link_libraries(blaise_options INTERFACE ws2_32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Haiku")
    target_link_libraries(blaise_options INTERFACE network)
endif()

set(core_sources)
foreach(name ${libraries})
    file(GLOB sources CONFIGURE_DEPENDS "lib/${name}/*.cpp")
    list(APPEND core_sources ${sources})
endforeach()
file(GLOB app_sources CONFIGURE_DEPENDS src/app/*.cpp)
add_library(blaise_core STATIC ${core_sources} ${app_sources})
target_link_libraries(blaise_core PUBLIC blaise_options)

file(GLOB ui_sources CONFIGURE_DEPENDS src/ui/*.cpp)
# Cross SDKs consume host-generated C++; never execute a target-built rc.
set(BLAISE_RESOURCE_CPP "" CACHE FILEPATH "Resource C++ generated by a host Native rc")
if(BLAISE_RESOURCE_CPP)
    if(NOT EXISTS "${BLAISE_RESOURCE_CPP}")
        message(FATAL_ERROR "BLAISE_RESOURCE_CPP does not exist")
    endif()
    set(resource_cpp "${BLAISE_RESOURCE_CPP}")
elseif(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Cross builds require BLAISE_RESOURCE_CPP; use scripts/build/platforms.py or generate it with make resources")
else()
    set(resource_cpp "${CMAKE_BINARY_DIR}/resources/blaise.cpp")
    file(GLOB resource_images CONFIGURE_DEPENDS src/resources/icons/*/*.png)
    add_custom_command(OUTPUT "${resource_cpp}"
        BYPRODUCTS "${CMAKE_BINARY_DIR}/resources/blaise.resources"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/resources"
        COMMAND $<TARGET_FILE:native_rc>
            "${CMAKE_SOURCE_DIR}/src/resources/blaise.json"
            -o "${CMAKE_BINARY_DIR}/resources/blaise.resources"
            --cpp "${resource_cpp}"
        DEPENDS native_rc src/resources/blaise.json ${resource_images}
        VERBATIM)
endif()
add_custom_target(blaise_resources DEPENDS "${resource_cpp}")
add_library(blaise_ui STATIC ${ui_sources} "${resource_cpp}")
target_include_directories(blaise_ui PUBLIC src/ui)
target_link_libraries(blaise_ui PUBLIC blaise_core native)

add_executable(blaise WIN32 src/main.cpp)
target_link_libraries(blaise PRIVATE blaise_ui)

file(GLOB ui_tests CONFIGURE_DEPENDS tests/ui/test_*.cpp)
add_executable(test_desktop ${ui_tests})
target_include_directories(test_desktop PRIVATE tests/unit)
target_compile_definitions(test_desktop PRIVATE
    FIXTURES="${CMAKE_SOURCE_DIR}/tests/fixtures")
target_link_libraries(test_desktop PRIVATE blaise_ui)
